// --- NUEVA LÓGICA DE PROGRESIÓN DE SKINS --- // Objeto con los requisitos de cada skin (puedes ampliarlo a 20) const requisitosSkins = { 'original': 0, 'mario': 5, 'cloud': 15, 'link': 50, 'sephiroth': 100 }; function simularAnuncio(idSkin, color, img) { // 1. Mirar cuántos anuncios lleva acumulados esta skin let anunciosVistos = parseInt(localStorage.getItem(`ads_${idSkin}`)) || 0; let objetivo = requisitosSkins[idSkin]; // 2. Si ya está desbloqueada, aplicarla directamente if (anunciosVistos >= objetivo) { aplicarSkin(idSkin, color, img); toggleSkins(); return; } // 3. Si no, pedir ver anuncio if(confirm(`¿Ver un anuncio para desbloquear? (${anunciosVistos}/${objetivo} vistos)`)) { const footer = document.getElementById('info-footer'); bloqueoFooter = true; footer.innerText = "📺 REPRODUCIENDO ANUNCIO..."; footer.style.background = "#f39c12"; setTimeout(() => { // Incrementar contador anunciosVistos++; localStorage.setItem(`ads_${idSkin}`, anunciosVistos); bloqueoFooter = false; if (anunciosVistos >= objetivo) { // ¡DESBLOQUEADO! aplicarSkin(idSkin, color, img); mostrarFeedback(`✨ ¡NUEVO ASPECTO DESBLOQUEADO!`, color); toggleSkins(); } else { // FALTA PROGRESO mostrarFeedback(`📺 ANUNCIO VISTO (${anunciosVistos}/${objetivo})`, "#3498db"); } }, 4000); // 4 segundos de "simulación" } }
